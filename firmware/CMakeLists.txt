cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2 CACHE STRING "Board")
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(rp_extio C CXX ASM)
pico_sdk_init()
add_executable(rp_extio main.c usb_descriptors.c)
target_include_directories(rp_extio PRIVATE . ..)
target_link_libraries(rp_extio pico_stdlib hardware_i2c hardware_spi hardware_uart tinyusb_device tinyusb_board)
pico_enable_stdio_usb(rp_extio 0)
pico_enable_stdio_uart(rp_extio 0)
pico_add_extra_outputs(rp_extio)
